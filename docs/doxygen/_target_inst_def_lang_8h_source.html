214 static constexpr auto value = lexy::callback<Ast::TargetInstDef::TargetInstDecl>(
215 [](Ast::Common::Identifier name,
216 std::pmr::vector<Ast::TargetInstDef::TargetOperandDecl> operands,
217 std::pmr::vector<BodyItem::ItemVariant> bodyItems,
218 auto...)
219 {
220 Ast::TargetInstDef::TargetInstDecl decl;
221 decl.m_instName = std::move(name);
222 decl.m_operands = std::move(operands);
223
224 for (auto &item : bodyItems)
225 {
226 std::visit(
227 [&](auto &&val)
228 {
229 using T = std::decay_t<decltype(val)>;
230 if constexpr (std::is_same_v<T, BodyItem::TagMnemonic>)
231 decl.m_mnemonic = std::move(val.val);
232 else if constexpr (std::is_same_v<T, BodyItem::TagFlags>)
233 decl.m_flags = std::move(val.val);
234 else if constexpr (std::is_same_v<T, BodyItem::TagImplicitDefs>)
235 decl.m_implicitDefs = std::move(val.val);
236 else if constexpr (std::is_same_v<T, BodyItem::TagImplicitUses>)
237 decl.m_implicitUses = std::move(val.val);
238 else if constexpr (std::is_same_v<T, BodyItem::TagEncoding>)
239 decl.m_encoding = std::move(val.val);
240 },
241 item);
242 }
243 return decl;
244 });
