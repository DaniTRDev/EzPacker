54 virtual LegalityResponse query(const LegalityQuery &q) const
55 {
56 size_t opIdx = static_cast<size_t>(q.m_opcode);
57 if (opIdx >= OPCODE_COUNT)
58 {
59 return LegalityResponse{ .m_action = LegalizeActionKind::Unsupported };
60 }
61
62 // Tier 3 / Wildcard check: unconditional actions (e.g. CALL, RET lowering)
63 if (m_wildcardActions[opIdx].m_action != LegalizeActionKind::Unsupported)
64 {
65 return m_wildcardActions[opIdx];
66 }
67
68 // Tier 2: Heterogeneous Rule Matchers
69 auto it = m_ruleMatchers.find(q.m_opcode);
70 if (it != m_ruleMatchers.end())
71 {
72 for (const auto &matcher : it->second)
73 {
74 LegalityResponse resp = matcher(q);
75 if (resp.m_action != LegalizeActionKind::Unsupported)
76 {
77 return resp;
78 }
79 }
80 }
81
82 // Tier 1: Dense 2D Primary Matrix (O(1))
83 if (q.m_operandCount > 0 && q.m_compactIds[0] < MAX_COMPACT_TYPES)
84 {
85 LegalityResponse fastResp = m_primaryMatrix[opIdx][q.m_compactIds[0]];
86 if (fastResp.m_action != LegalizeActionKind::Unsupported)
87 {
88 return fastResp;
89 }
90 }
91
92 return LegalityResponse{ .m_action = LegalizeActionKind::Unsupported };
93 }

470 void installScalarAction(size_t matchSlot,
471 size_t actionSlot,
472 std::initializer_list<MirType *> fromTypes,
473 MirType *toType,
474 LegalizeActionKind action)
475 {
476 uint8_t toId = toType ? toType->getCompactId() : 0;
477 for (auto op : m_opcodes)
478 {
479 if (matchSlot == 0 && actionSlot == 0)
480 {
481 for (auto *t : fromTypes)
482 {
483 if (t && t->getCompactId() < LegalizerInfo::MAX_COMPACT_TYPES)
484 {
485 m_info->setPrimaryMatrix(op,
486 t->getCompactId(),
487 LegalityResponse{ .m_action = action,
488 .m_slot = 0,
489 .m_targetCompactId = toId });
490 }
491 }
492 continue;
493 }
494
495 std::vector<uint8_t> fromIds;
496 for (auto *t : fromTypes)
497 {
498 if (t)
499 fromIds.push_back(t->getCompactId());
500 }
501 m_info->addRuleMatcher(
502 op,
503 [matchSlot, actionSlot, fromIds, toId, action](const LegalityQuery &q) -> LegalityResponse
504 {
505 if (q.m_operandCount > matchSlot)
506 {
507 for (uint8_t fid : fromIds)
508 {
509 if (q.m_compactIds[matchSlot] == fid)
510 {
511 return LegalityResponse{ .m_action = action,
512 .m_slot = static_cast<uint8_t>(actionSlot),
513 .m_targetCompactId = toId };
514 }
515 }
516 }
517 return LegalityResponse{ .m_action = LegalizeActionKind::Unsupported };
518 });
519 }
520 }
