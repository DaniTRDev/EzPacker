58struct AggregateCondition
59{
60 enum class Kind : uint8_t
61 {
62 NonTrivial, // Aggregate has a non-trivial copy/destructor.
63 Unaligned, // Aggregate's alignment is not naturally satisfied.
64 SizeGreaterThan, // Aggregate size exceeds m_sizeLimit.
65 SizeLessThanOrEqual, // Aggregate size is at most m_sizeLimit.
66 SizeIn, // Aggregate size is one of m_sizeSet.
67 HomogeneousAggregate, // e.g. HFA/HVA
68 Default // Fallback applied when no earlier condition matched.
69 } m_kind;
70
71 std::optional<Common::IntegerLiteral> m_sizeLimit; // Threshold used by the size comparison kinds.
72 std::pmr::vector<Common::IntegerLiteral> m_sizeSet; // Explicit accepted sizes for SizeIn.
73 std::optional<Common::Identifier> m_homoBaseType; // e.g. "float"
74 std::optional<Common::IntegerLiteral> m_homoMaxCount; // e.g. 4
75
76 Common::Identifier m_resultClass; // e.g. "integer", "by_ref", "memory"
77 bool m_implicitCopy{ false }; // True for Win64-style by_ref stack copy
78};
